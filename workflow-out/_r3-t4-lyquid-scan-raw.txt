#!/usr/bin/env python3
# T4.5 Lyquid 只读取数：扫描两实例在共享链上的全部交易（含 to=null 的创建交易）。
# 只调用 eth_blockNumber / eth_getBlockByNumber / eth_getTransactionReceipt；不发交易。
# 数据源只读；endpoint 为 JSON-RPC 的 /api 路径（根路径 404，T4.5 现场实证）。
import json
import sys
import urllib.request

EP = "http://node1:10087/api"
R3 = "0x87006e75a5b6be9d1bbf61ac8cd84f05d9140589"
R2 = "0xfd6f7a6a5c21a3f503ebae7a473639974379c351"
INST = {R3: "R3", R2: "R2"}

# selector -> 方法名（来源：workflow-out/w1b2-s4-rpc.md:66-79 固化的 cast sig 表；
# apply_committee_verdict 由容器内 `cast sig "apply_committee_verdict(string,uint8,string[])"` 实算并交叉校验）
SEL = {
    "0xb540def8": "faucet(string,uint128)",
    "0xbd827024": "create_deal(string,string,uint128,uint16,string,string,uint64,uint8)",
    "0x73451f65": "fund(string)",
    "0xb1fecd5a": "notarize(string,string,string,string)",
    "0x7ecc9d46": "submit_deliverable(string,string)",
    "0xad932b5c": "complete(string)",
    "0x786cfc8f": "claim_timeout(string)",
    "0x0b4f3f3d": "cancel(string)",
    "0xa48ab911": "open_dispute(string,string)",
    "0x15df3c73": "committee_config(string[],uint32)",
    "0xec78f86e": "deal_info(string)",
    "0x16417010": "evidence_entries_json(string)",
    "0x3e6b4459": "balance_of(string)",
    "0x4704839a": "due_deals(uint64)",
    "0xa32974e3": "apply_committee_verdict(string,uint8,string[])",
    "0x": "(empty calldata / plain transfer)",
}


def rpc(method, params):
    payload = json.dumps({"jsonrpc": "2.0", "id": 1, "method": method, "params": params}).encode()
    req = urllib.request.Request(EP, data=payload, headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=60) as f:
        r = json.load(f)
    if "error" in r:
        raise RuntimeError("%s -> %s" % (method, r["error"]))
    return r["result"]


head = int(rpc("eth_blockNumber", []), 16)
print("endpoint=%s  eth_blockNumber=%s (dec %d)" % (EP, hex(head), head))
print("scan_range=0..%d (inclusive)  instances: R3=%s R2=%s" % (head, R3, R2))

rows = []
creations = []
null_to_seen = 0
for n in range(0, head + 1):
    b = rpc("eth_getBlockByNumber", [hex(n), True])
    if not b:
        print("WARN block %d returned null" % n, file=sys.stderr)
        continue
    txs = b.get("transactions") or []
    for tx in txs:
        to = tx.get("to")
        h = tx["hash"]
        sel = (tx.get("input") or "0x")[:10].lower()
        if to in (None, ""):
            null_to_seen += 1
            rc = rpc("eth_getTransactionReceipt", [h])
            ca = (rc.get("contractAddress") or "").lower()
            if ca in INST:
                creations.append({
                    "inst": INST[ca], "block": n, "hash": h, "frm": tx.get("from", ""),
                    "ca": ca, "gas": int(rc.get("gasUsed") or "0x0", 16),
                    "status": rc.get("status"), "sel": sel,
                    "method": SEL.get(sel, "unknown(%s)" % sel),
                })
        elif to.lower() in INST:
            rc = rpc("eth_getTransactionReceipt", [h])
            rows.append({
                "inst": INST[to.lower()], "block": n, "hash": h, "frm": tx.get("from", ""),
                "to": to, "gas": int(rc.get("gasUsed") or "0x0", 16),
                "status": rc.get("status"), "sel": sel,
                "method": SEL.get(sel, "unknown(%s)" % sel),
            })

print("null_to_txs_total=%d  creation_txs_to_instances=%d  instance_calls=%d"
      % (null_to_seen, len(creations), len(rows)))
print()
print("===== 创建交易（to=null，receipt.contractAddress ∈ {R2,R3}）=====")
print("inst\tblock\thash\tfrom\tcontractAddress\tgasUsed\thex\tstatus\tselector\tmethod")
for c in sorted(creations, key=lambda x: x["block"]):
    print("%s\t%d\t%s\t%s\t%s\t%d\t0x%x\t%s\t%s\t%s"
          % (c["inst"], c["block"], c["hash"], c["frm"], c["ca"], c["gas"], c["gas"], c["status"], c["sel"], c["method"]))
print()
print("===== 实例地址交易（to ∈ {R2,R3}）=====")
print("inst\tblock\thash\tfrom\tgasUsed\thex\tstatus\tselector\tmethod")
for r in sorted(rows, key=lambda x: x["block"]):
    print("%s\t%d\t%s\t%s\t%d\t0x%x\t%s\t%s\t%s"
          % (r["inst"], r["block"], r["hash"], r["frm"], r["gas"], r["gas"], r["status"], r["sel"], r["method"]))
print()
print("===== 汇总 =====")
for name, key in (("R2", "R2"), ("R3", "R3")):
    cr = [c for c in creations if c["inst"] == key]
    rs = [r for r in rows if r["inst"] == key]
    allx = cr + rs
    gas = sum(x["gas"] for x in allx)
    blocks = sorted(set(x["block"] for x in allx))
    print("%s: 创建交易 %d 笔；调用交易 %d 笔；合计 %d 笔；gasUsed 合计 %d (0x%x)；块范围 %s..%s"
          % (name, len(cr), len(rs), len(allx), gas, gas,
             blocks[0] if blocks else "-", blocks[-1] if blocks else "-"))
    if cr:
        dep = cr[0]
        print("   %s 部署: block %d tx %s gasUsed %d (0x%x) contractAddress %s"
              % (name, dep["block"], dep["hash"], dep["gas"], dep["gas"], dep["ca"]))
        if rs:
            last = max(rs, key=lambda x: x["block"])
            print("   %s 部署→最后一笔实例交易块跨度: %d -> %d = %d 块（最后一笔 block %d tx %s %s）"
                  % (name, dep["block"], last["block"], last["block"] - dep["block"],
                     last["block"], last["hash"], last["method"]))
print()
print("===== per-instance 方法分布 =====")
for name in ("R2", "R3"):
    print("[%s]" % name)
    counts = {}
    for x in rows:
        if x["inst"] == name:
            counts.setdefault(x["method"], []).append(x["block"])
    for m in sorted(counts):
        bs = counts[m]
        print("  %-62s n=%d blocks=%s" % (m, len(bs), ",".join(str(i) for i in bs)))
